53Unique_id_holder::id_t Unique_id_holder::create_unique_id() // Static.
54{
55 using std::atomic;
56 using std::memory_order_relaxed;
57
58 /* Get the next available ID in the universe. Use large type for id_t to avoid overflow.
59 * Use atomic<> to ensure thread safety. Use pre-increment ++x equivalent, so that the smallest possible ID 0 is
60 * reserved and never issued (might come in handy for debugging someday, identifying failure of
61 * this feature).
62 *
63 * Use local static; in C++17 this is formally thread-safe; this way the init happens on-demand (no danger of
64 * static-init mis-ordering). */
65 static atomic<id_t> s_last_id{0};
66 return s_last_id.fetch_add(1, memory_order_relaxed) + 1;
67}

71Thread_token this_thread_unique_token()
72{
73 static_assert(std::is_same_v<uint64_t, Unique_id_holder::id_t>,
74 "Thread_token wraps Unique_id_holder-supplied values; this check ensures no surprises: that "
75 "Unique_id_holder issues the type we expect.");
76
77 /* Unique_id_holder::create_unique_id() never returns 0 (its first result is 1); hence default-cted
78 * Thread_token (= 0 inside) = not-yet-assigned, as Thread_token doc header promises.
79 * Both `static` (inside create_unique_id()) and this `thread_local` are trivially destructible and
80 * constant-initialized => usable at any time including during static/TLS teardown as advertised. */
81 thread_local Thread_token s_token;
82 if (s_token == Thread_token{})
83 {
84 s_token = Thread_token{Unique_id_holder::create_unique_id()}; // Slow-path.
85 }
86 // else { Fast-path: no-op. }
87 return s_token;
88}

90std::ostream& operator<<(std::ostream& os, Thread_token val)
91{
92 using boost::io::ios_all_saver;
93 using std::hex;
94 using std::setfill;
95 using std::setw;
96
97 ios_all_saver saver{os}; // Restore formatting at end of {block}.
98
99 /* Print the low 32 bits only (cf. abbreviated git hashes): the value comes from a monotonic counter, so
100 * the high bits are zeros until the 2^32nd ID is issued process-wide -- i.e., in practice always; printing
101 * them would be pure noise. (Display-truncation only: the stored value keeps all 64 bits, and code
102 * needing the full value printed can cast: `os << Thread_token::id_t(val)`.) */
103 static_assert(sizeof(uint32_t) == 4, "4 bytes <=> 8 hex-chars.");
104 return os << "0x" << hex << setfill('0') << setw(8) << uint32_t(Thread_token::id_t(val));
105}
