36/**
37 * An empty interface, consisting of nothing but a default `virtual` destructor, intended as a boiler-plate-reducing
38 * base for any other (presumably `virtual`-method-having) class that would otherwise require a default `virtual`
39 * destructor.
40 *
41 * Usually, if you have a base class `C` at the top of a `virtual`-method-having hierarchy, then it needs a `virtual`
42 * destructor, even if it is `= default` or `{}`. Otherwise, trying to delete an object of subclass `C2 : public C`
43 * via a `C*` pointer will fail to call destructor `~C2()` -- which may not be empty, causing leaks and so on.
44 * Declaring `~C()` and its empty implementation is surprisingly verbose. So, instead, don't; and `public`ly derive
45 * from Null_interface.
46 *
47 * It is particularly useful for interface classes.
48 */
49class Null_interface
50{
51public:
52 // Destructor.
53
54 /**
55 * Boring `virtual` destructor.
56 *
57 * Why is it pure? Main reason: Then Null_interface becomes abstract and cannot be itself instantiated, which is good.
58 * Otherwise we'd need a `protected` constructor or something to prevent it.
59 *
60 * I (ygoldfel) actually thought this means a subclass *has* to now define a body (even if merely `= default` or
61 * `{}`): but no: the compiler will (and must) generate an empty (and, because of us, `virtual`) destructor for any
62 * subclass that doesn't explicitly define one. A destructor isn't a regular method, so that's how it works.
63 * There will not be a linker error.
64 */
65 virtual ~Null_interface() = 0;
66};

93/**
94 * A simple RAII-pattern class template that, at construction, sets the specified location in memory to a specified
95 * value, memorizing the previous contents; and at destruction restores the value. E.g.:
96 *
97 * ~~~
98 * thread_local int s_this_thread_val;
99 * ...
100 * {
101 * Scoped_setter<int> setter{&s_this_thread_val, 75}; // Set it to 75 and memorize (whatever).
102 * ...
103 * {
104 * Scoped_setter<int> setter{&s_this_thread_val, 125}; // Set it to 125 and memorize 75.
105 * ...
106 * } // Restored from 125 to 75.
107 * ...
108 * } // Restored from (probably) 75 to (whatever).
109 * ~~~
110 *
111 * The object is movable, not copyable (which is similar to `unique_ptr`) to prevent "double-restore." Related:
112 * one can easily return customized auto-setter/restorers:
113 *
114 * ~~~
115 * thread_local Widget s_widget;
116 * auto widget_setter_auto(Widget&& widget_moved)
117 * {
118 * return flow::util::Scoped_setter<Widget>{&s_widget, std::move(widget_moved)};
119 * }
120 * ...
121 * { // Later, some block: Set s_widget. Code here doesn't even know/care a Scoped_setter is involved.
122 * const auto setter_auto = widget_setter_auto({ ...widget-init... });
123 * ...
124 * } // Restore s_widget.
125 * ~~~
126 *
127 * ### Thready safety ###
128 * This is a simple object: it just performs a few assignments without any added
129 * concurrency protection. If the memory location can be accessed simultaneously by other threads, watch out.
130 *
131 * In particular it's a good fit for thread-local locations: `&X`, where `X` is declared `thread_local`, or
132 * `X == *(P.get())` where `P` is a `boost::thread_specific_ptr` or equivalent.
133 *
134 * @tparam Value_t
135 * The stored type, which must be move-assignable and move-constructible.
136 * All `Value` writes are performed using exclusively these operations.
137 * Informally: For best performance when `Value` is a heavy-weight type, these operations should be
138 * be written to be light-weight, such as in terms of swapping a few scalars.
139 * In particular this is already the case for all STL-compliant container types.
140 *
141 * @internal
142 * ### Implementation ###
143 * An alternative implementation, which could even be reduced to just an alias, would have used `unique_ptr`.
144 * However in this case I (ygoldfel) wanted maximum control for perf. The use case originally driving this was
145 * the thread-local verbosity override: log::Config::this_thread_verbosity_override_auto(). flow::log is fairly
146 * paranoid about performance, in general, although admittedly this particular call isn't necessarily ubiquitous.
147 */
148template<typename Value_t>
149class Scoped_setter
150{
151public:
152 // Types.
153
154 /// Alias for template parameter.
155 using Value = Value_t;
156
157 // Constructors/destructor.
158
159 /**
160 * Post-condition: `*target` contains was `val_src_moved` contained at ctor entry; and the destructor invocation shall
161 * reverse this, so that `*target` is restored to its value at entry.
162 *
163 * `*this` cannot be copied, but it can be moved. As a result, it is guaranteed that the aforementioned destructor
164 * will execute exactly once; however it can be move-constructed-from-`*this` other Scope_setter's destructor,
165 * while our own dtor therefore is a no-op.
166 *
167 * @param target
168 * The target object that shall be set to `val_src_moved` now and restored in our, or moved-object's, dtor.
169 * The current value of `*target` is saved internally via assignment of `move(*target)`.
170 * Behavior undefined (assertion may trip) if null.
171 * @param val_src_moved
172 * Value to save to `*target` immediately, via assignment of `move(val_src_moved)`.
173 */
174 explicit Scoped_setter(Value* target, Value&& val_src_moved);
175
176 /**
177 * Move constructor: `*this` acts as `src_moved` would-have, while `src_moved` becomes a no-op object permanently.
178 *
179 * @param src_moved
180 * Source object. Its destructor shall do nothing after this returns.
181 */
182 Scoped_setter(Scoped_setter&& src_moved);
183
184 /// Prohibit copying: for each `explicit` ctor invocation, there shall be exactly 1 non-no-op dtor invocation.
185 Scoped_setter(const Scoped_setter&) = delete;
186
187 /**
188 * Restores `*target` (from main ctor) to its value at entry to said ctor; or does nothing if `*this` has been
189 * moved-from via the move ctor.
190 */
191 ~Scoped_setter();
192
193 // Methods.
194
195 /// Prohibit copying (see `delete`d copy ctor).
196 Scoped_setter& operator=(const Scoped_setter&) = delete;
197
198 /// Prohibit modifying existing `*this`; except that moving-from is enabled via the move ctor.
199 Scoped_setter& operator=(Scoped_setter&&) = delete;
200
201private:
202 // Data.
203
204 /// Target object location; see ctors; if null then this is a moved-from Scoped_setter that intentionally no-ops.
205 Value* m_target_or_null;
206
207 /// If and only if #m_target_or_null is non-null, this saves `*m_target_or_null`. Otherwise meaningless.
208 Value m_saved_value;
209}; // class Scoped_setter

211/**
212 * A simple locking-proxy class template (execute-around-pointer idiom): at construction locks the given mutex and
213 * provides `->` access to the given target object; at destruction unlocks it. In typical use one obtains `*this` as
214 * an unnamed temporary in the middle of an expression, so that the mutex is locked for exactly that expression's
215 * duration. E.g., given `Widget w` and a mutex `m` serializing all access to `w` -- suppose some `w_locked()` returns
216 * `Locked_proxy{&w, &m}`:
217 *
218 * ~~~
219 * const auto n = w_locked()->compute_n(); // Lock m; invoke w.compute_n(); save the result; unlock m.
220 * ~~~
221 *
222 * This is the *execute-around-pointer idiom*. The typical motivation: some class `C` internally owns both a
223 * mutex-protected object and the mutex, its background/concurrent machinery locking the latter as needed; and
224 * wants to let its user safely invoke methods of that object. Then `C` can supply a `w_locked()`-style accessor
225 * as sketched above -- instead of writing a delegating, internally-locking method for every operation of
226 * interest. The set of available operations then maintains itself, as does the correctness of each such call's
227 * locking.
228 *
229 * To grant read-only (`const` methods only) access, simply use a `const`-qualified #Target type: e.g.,
230 * `Locked_proxy<const Widget, ...>`.
231 *
232 * ### Keep it to one expression ###
233 * `*this` existing = the mutex being locked. The intended use is as an unnamed temporary (see above example):
234 * then the lock is held for a few microseconds, and it is impossible to leak. If you *do* save a `*this` to
235 * extend the locked section, keep the scope tight: whatever machinery normally locks that mutex is blocked
236 * in the meantime.
237 *
238 * @see `boost::synchronized_value` implements the same idiom but *owns* both the payload and the mutex
239 * ("wrap your data in me"). `*this`, by contrast, is non-owning: it is for objects whose guarding
240 * mutex exists independently -- e.g., one that guards more state than just the target object.
241 *
242 * @tparam Target_t
243 * Type of the object to which `*this` proxies access; possibly `const`-qualified (see above).
244 * @tparam Mutex_t
245 * A `Lockable`-concept mutex type, e.g. util::Mutex_non_recursive.
246 */
247template<typename Target_t, typename Mutex_t>
248class Locked_proxy
249{
250public:
251 // Types.
252
253 /// Alias for template parameter.
254 using Target = Target_t;
255
256 /// Alias for template parameter.
257 using Mutex = Mutex_t;
258
259 // Constructors/destructor.
260
261 /**
262 * Locks `*mutex` -- blocking as-needed until that is possible -- and makes `*this` proxy `*target`.
263 * The destructor invocation shall unlock it.
264 *
265 * `*this` cannot be copied, but it can be moved. As a result, it is guaranteed that the aforementioned
266 * unlocking will occur exactly once; however it can occur via the destructor of another Locked_proxy that was
267 * move-constructed from `*this`, our own dtor therefore being a no-op. (A moved-from `*this` must not be
268 * dereferenced.)
269 *
270 * @param target
271 * The object to which operator->() shall forward. Behavior undefined (assertion may trip) if null.
272 * @param mutex
273 * The mutex to hold for `*this` lifetime: the mutex -- the one and only one -- by which all
274 * access to `*target` is serialized (see class doc header for the typical setup).
275 * Must be non-null; must outlive `*this`.
276 */
277 explicit Locked_proxy(Target* target, Mutex* mutex);
278
279 /**
280 * Move constructor: `*this` acts as `src_moved` would-have (in particular its eventual destruction unlocks the
281 * mutex), while `src_moved` becomes a no-op object permanently.
282 *
283 * @param src_moved
284 * Source object. Its destructor shall do nothing after this returns; it must not be dereferenced.
285 */
286 Locked_proxy(Locked_proxy&& src_moved);
287
288 /// Prohibit copying: for each `explicit` ctor invocation, there shall be exactly 1 unlocking dtor invocation.
289 Locked_proxy(const Locked_proxy&) = delete;
290
291 // Methods.
292
293 /// Prohibit copying (see `delete`d copy ctor).
294 Locked_proxy& operator=(const Locked_proxy&) = delete;
295
296 /// Prohibit modifying existing `*this`; except that moving-from is enabled via the move ctor.
297 Locked_proxy& operator=(Locked_proxy&&) = delete;
298
299 /**
300 * Returns pointer to the target object, the guarding mutex being locked by `*this` (do not use if moved-from).
301 * @return See above.
302 */
303 Target* operator->() const;
304
305 /**
306 * Returns reference to the target object, the guarding mutex being locked by `*this` (do not use if moved-from).
307 * Useful, e.g., to print the target (`os << *w_locked()`) or to pass it to a function taking a reference --
308 * keeping in mind the mutex is locked only until the end of the full expression (in typical unnamed-temporary
309 * use).
310 *
311 * @return See above.
312 */
313 Target& operator*() const;
314
315private:
316 // Data.
317
318 /// Holds the caller-supplied mutex locked throughout `*this` lifetime (unless moved-from: then holds nothing).
319 Lock_guard<Mutex> m_lock;
320
321 /// The target object to which operator->() forwards. Meaningless if `*this` is moved-from.
322 Target* m_target;
323}; // class Locked_proxy

337template<typename Value_t>
338Scoped_setter<Value_t>::Scoped_setter(Scoped_setter&& src_moved) : // =default might work fine but to be clear/certain:
339 m_target_or_null(src_moved.m_target_or_null),
340 m_saved_value(std::move(src_moved.m_saved_value))
341{
342 assert(m_target_or_null && "Should not be moving-from a thing that has already been moved-from.");
343
344 src_moved.m_target_or_null = nullptr;
345 // As promised: Now src_moved's dtor will no-op.
346}

348template<typename Value_t>
349Scoped_setter<Value_t>::~Scoped_setter()
350{
351 if (m_target_or_null)
352 {
353 *m_target_or_null = std::move(m_saved_value);
354 }
355 // else { `*this` must have been moved-from. No-op. }
356}

385template<typename Time_unit, typename N_items>
386double to_mbit_per_sec(N_items items_per_time, size_t bits_per_item)
387{
388 /* Let there be U/W seconds per Time_unit. Then the following holds:
389 *
390 * items_per_time items/Time_units * W/U Time_units/second * bits_per_item bits/item
391 * * 1/(1000*1000) megabits/bits
392 * = items_per_time * W / U * bits_per_item / 1000 / 1000 megabits/second.
393 *
394 * Notice the units work out. W and U are conveniently available in Time_unit::period, which is a boost::ratio. */
395
396 return
397 /* This zealously converts everything to double ASAP to avoid overflow. Could probably speed things up a bit
398 * by postponing some of those conversions until after some integer multiplications, but then overflows could
399 * creep in. It's best not to assume too much about the values of den and num, as this function is meant to
400 * be rather adaptable to various situations. I did try to avoid unnecessary divisions though in favor of
401 * multiplications, sort of guessing the latter are faster. Or not... *shrug*. */
402 double(items_per_time) * double(bits_per_item) * double(Time_unit::period::den)
403 / (double(Time_unit::period::num) * double(1000 * 1000));
404}

406template<typename Integer, typename Integer2>
407constexpr Integer ceil_div(Integer dividend, Integer2 divisor)
408{
409 // ceil(A : B) = (A + B - 1) / B, where : is floating point division, while / is integer division.
410 static_assert(std::is_integral_v<Integer>, "ceil_div<T, T2>: T must be an integer type.");
411 static_assert(std::is_integral_v<Integer2>, "ceil_div<T, T2>: T2 must be an integer type.");
412
413 // assert(dividend >= 0); // Cannot do that (or throw in C++17, can in C++20; @todo) in constexpr.
414 // assert(divisor > 0); // Ditto.
415
416 return (dividend + static_cast<Integer>(divisor) - static_cast<Integer>(1)) / static_cast<Integer>(divisor);
417 /* (Could one do further bitwise trickery? Perhaps but let optimizer do it. Wouldn't optimizer also just
418 * optimize a literal floating-point `ceil(a / b)`? Well, no. Probably not. So we wrote this function.) */
419}

421template<typename Integer, typename Integer2>
422constexpr Integer round_to_multiple(Integer dividend, Integer2 unit)
423{
424 return static_cast<Integer>(unit) * ceil_div(dividend, unit);
425}

480template<typename T>
481bool in_closed_range(T const & min_val, T const & val, T const & max_val)
482{
483 // This writes "(min_val <= val) && (val <= max_val)" by using only <, to support the greatest number of types.
484 return ((min_val < val) || (!(val < min_val))) &&
485 ((val < max_val) || (!(max_val < val)));
486}

526template<typename Minuend, typename Subtrahend>
527bool subtract_with_floor(Minuend* minuend, const Subtrahend& subtrahend, const Minuend& floor)
528{
529 assert(minuend);
530
531 /* Basically just avoid implicit conversions and anything that mind overflow or underflow.
532 * The one underflow we allow is the subtraction of `floor`: doc header says keep `floor` small.
533 * So it's their problem if it's not. */
534
535 const Minuend converted_subtrahend = Minuend{subtrahend};
536
537 // min - sub <= floor <===> min - floor <= sub.
538 if ((*minuend - floor) <= converted_subtrahend)
539 {
540 *minuend = floor;
541 return false;
542 }
543 // else
544 *minuend -= converted_subtrahend;
545 return true;
546}

560template<typename... T>
561void ostream_op_to_string(std::string* target_str, T&&... ostream_args)
562{
563 using std::flush;
564
565 /* Pushes characters directly onto an `std::string`, instead of doing so into an `ostringstream` and then getting it
566 * by copy via `ostringstream::copy()`. This is for performance and may make a large difference
567 * overall, if this is used in logging for example. However, Thread_local_string_appender accomplishes
568 * better performance still and some other features. */
569 String_ostream os{target_str};
570 feed_args_to_ostream(&(os.os()), std::forward<T>(ostream_args)...);
571 os.os() << flush;
572}

610 size_t bytes_per_line)
611{
612 using boost::io::ios_fill_saver;
613 using boost::io::ios_flags_saver;
614 using boost::io::ios_width_saver;
615 using boost::asio::buffers_iterator;
616 using std::isprint;
617
618 /* This sweet type will iterate over the buffer sequence (jumping between contained buffers, if ther are > 1).
619 * If `Bufs_iter it`, then *it is of type uint8_t. */
620 using Bufs_iter = buffers_iterator<Const_buffer_sequence, uint8_t>;
621
622 constexpr size_t BYTES_PER_LINE_DEFAULT = 16;
623 bool single_line_mode = false;
624 if (bytes_per_line == 0)
625 {
626 bytes_per_line = BYTES_PER_LINE_DEFAULT;
627 }
628 else if (bytes_per_line == size_t(-1))
629 {
630 /* Firstly just pretend exactly the bytes in the whole input = "max" bytes per line.
631 * This accomplishes the bulk of what "single-line mode" means. */
632 bytes_per_line = buffer_size(data);
633 // A bit slow potentially to need to enumerate all scattered buffers. Eh. Contract said we should be assumed slow.
634
635 // The rest of what it means is lacking a newline at the end in single-line mode. So just remember that part.
636 single_line_mode = true;
637 }
638
639 // Ensure format settings return to their previous values subsequently.
640 ios_flags_saver flags_saver{os};
641 ios_fill_saver fill_saver{os};
642 ios_width_saver width_saver{os};
643
644 /* Set formatting and output numeric value (hex) of first byte.
645 * @todo Is there a way to write this with manipulators too? */
646 os.setf(std::ios::right | std::ios::hex, std::ios::adjustfield | std::ios::basefield);
647 os << std::setfill('0');
648
649 const Bufs_iter end_byte_it = Bufs_iter::end(data);
650 for (Bufs_iter cur_byte_it = Bufs_iter::begin(data);
651 cur_byte_it != end_byte_it;
652 /* Advancing of cur_byte_it occurs within body of loop. */)
653 {
654 // The for() loop around us guarantees there is at least this first byte. Print the numeric value.
655 os << indentation << '['
656 << std::setw(2) << int(*cur_byte_it); // Numeric value in hex.
657
658 // Repeat for remaining bytes left in this line. Stop at bytes/line limit, or if reached end of buffers.
659 size_t n_bytes_printed;
660 for ((n_bytes_printed = 1), ++cur_byte_it; // Account for printing that first byte above.
661 (n_bytes_printed != bytes_per_line) && (cur_byte_it != end_byte_it);
662 ++cur_byte_it, ++n_bytes_printed)
663 {
664 os << ' ' << std::setw(2) << int(*cur_byte_it); // Numeric value in hex.
665 }
666
667 // Spaces as if rest of line still had a few ghost values to print (space + 2 spaces for the hex digits).
668 for (size_t n_bytes_printed_including_padding = n_bytes_printed;
669 n_bytes_printed_including_padding != bytes_per_line;
670 ++n_bytes_printed_including_padding)
671 {
672 os << " ";
673 }
674
675 // Backtrack and print those same bytes -- this time as printable characters (when printable, else dots).
676 cur_byte_it -= n_bytes_printed;
677
678 os << '|';
679 for (size_t n_chars_printed = 0;
680 n_chars_printed != n_bytes_printed;
681 ++cur_byte_it, ++n_chars_printed)
682 {
683 char c = *cur_byte_it;
684 os << (isprint(c) ? c : '.');
685 }
686 os << ']';
687 if (!single_line_mode)
688 {
689 os << '\n';
690 }
691 } // for (cur_byte_it)
692
693 return os;
694
695 // The formatting changes will be restored here as the savers exit scope.
696} // buffers_to_ostream()

715template<typename Enum>
716Enum istream_to_enum(std::istream* is_ptr, Enum enum_default, Enum enum_sentinel,
717 bool accept_num_encoding, bool case_sensitive,
718 Enum enum_lowest)
719{
720 using boost::lexical_cast;
721 using boost::bad_lexical_cast;
722 using boost::algorithm::equals;
723 using boost::algorithm::is_iequal;
724 using std::locale;
725 using std::string;
726 using std::isdigit;
727 using std::isalnum;
728 using Traits = std::char_traits<char>;
729 using enum_t = std::underlying_type_t<Enum>;
730
731 // Reminder: There are various assumptions about Enum this takes for granted; behavior undefined otherwise.
732
733 assert(enum_t(enum_lowest) >= 0); // Otherwise we'd have to allow '-' (minus sign), and we'd... just rather not.
734 auto& is = *is_ptr;
735 const is_iequal i_equal_func{locale::classic()};
736
737 // Read into `token` until (and not including) the first non-alphanumeric/underscore character or stream end.
738 string token;
739 char ch;
740 while (((ch = is.peek()) != Traits::eof()) && (isalnum(ch) || (ch == '_')))
741 {
742 token += ch;
743 is.get();
744 }
745
746 Enum val = enum_default;
747
748 if (!token.empty())
749 {
750 if (accept_num_encoding && isdigit(token.front())) // Hence ostream<< shouldn't serialize a digit-leading value.
751 {
752 enum_t num_enum;
753 try
754 {
755 num_enum = lexical_cast<enum_t>(token);
756 // This assumes a vanilla enum integer value ordering.
757 if ((num_enum >= enum_t(enum_sentinel) || (num_enum < enum_t(enum_lowest))))
758 {
759 num_enum = enum_t(enum_default);
760 }
761 val = Enum{num_enum};
762 }
763 catch (const bad_lexical_cast&)
764 {
765 assert(val == enum_default);
766 }
767 } // if (accept_num_encoding && isdigit())
768 else // if (!(accept_num_encoding && isdigit()))
769 {
770 enum_t idx;
771 // This assumes a vanilla enum integer value ordering within this [closed range].
772 for (idx = enum_t(enum_lowest); idx != enum_t(enum_sentinel); ++idx)
773 {
774 const auto candidate = Enum{idx};
775 /* Note -- lexical_cast<string>(Enum) == (operator<<(ostringstream&, Enum)).str() -- the symbolic
776 * encoding of Enum (as we promised to accept, case-[in]sensitively), not the numeric encoding. The numeric
777 * encoding is checked-for in the `if (accept_num_encoding...)` branch above using a non-looping technique. */
778 if (case_sensitive ? equals(token, lexical_cast<string>(candidate))
779 : equals(token, lexical_cast<string>(candidate), i_equal_func))
780 {
781 val = candidate;
782 break;
783 }
784 }
785 assert((idx != enum_t(enum_sentinel)) || (val == enum_default));
786 } // else if (!(accept_num_encoding && isdigit()))
787 } // if (!token.empty())
788
789 return val;
790} // istream_to_enum()
